int leds[] = {2 , 3,4,5,6};



void setup()
{
  pinMode( 2 , OUTPUT);
  pinMode( 3 , OUTPUT);
  pinMode( 4 , OUTPUT);
  pinMode( 5 , OUTPUT);
  pinMode( 6 , OUTPUT);
}

void loop()
{
  
  
  // i es igual a 0 , 
  // el for tiene que recorrer mientras que i sea menor que 5
  // i va en aumento
  for (int i = 0 ; i < 5 ; i++ ){
  
    digitalWrite(leds[i],HIGH);
    delay(1000);
    digitalWrite(leds[i],LOW);
    delay(1000);
  
  }
    
  
  
  
}
